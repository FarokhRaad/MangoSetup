# =============================================================================
#  Package manifest for mango + DankMaterialShell
# =============================================================================
#  Format:  one package per line. '#' starts a comment.
#           Lines ending in "  # optional" are skipped unless --with-optional
#           is passed to 10-packages.sh.
#
#  The installer routes each package to pacman or the AUR automatically and
#  skips anything already present, so this is safe to re-run.
#
#  Upstream naming traps worth knowing (all verified against pacman -Si and
#  the AUR RPC, since a wrong name here fails the whole install):
#    mangowc / mangowc-git   RENAMED. The project moved to the 'mangowm' org
#                            and both old AUR names now 404. Use 'mangowm'.
#    p7zip                   RENAMED upstream to '7zip'.
#    qt5ct / qt6ct           the '-kde' variants are used here instead; see
#                            the Qt theming section for why (they conflict,
#                            so do not list both).
#    xdg-users-dirs          common typo; the real package is xdg-user-dirs.
#    dgop-git                its PKGBUILD builds ./cmd/cli, but upstream
#                            renamed that path to cmd/dgop, so the git
#                            package FAILS to build. Use 'dgop-bin'.
# =============================================================================

mangowm

# -----------------------------------------------------------------------------
#  Shell (DankMaterialShell)
# -----------------------------------------------------------------------------
#  DMS is a Quickshell (QML) + Go desktop shell. It has genuine first-class
#  mango support, not a generic wlroots fallback: it ships a MangoService
#  that speaks mango's native JSON-over-Unix-socket IPC, and `dms setup
#  <fragment>` generates mango config fragments into ~/.config/mango/dms/
#  (colors, layout, cursor, outputs, windowrules, binds), which config.conf
#  sources last. Docs: https://danklinux.com/docs/
#
#  Package choice:
#    dms-shell       stable, in the official [extra] repo
#    dms-shell-git   AUR, tracks main
#  Listed as the git package to stay current with mango IPC changes; switch
#  to 'dms-shell' from [extra] if you prefer stable.
dms-shell-git

#  Quickshell is DMS's ONLY hard UI dependency (in [extra], no AUR fork
#  needed). DMS embeds its QML in the dms binary but still needs the runtime.
quickshell

#  dgop is DMS's system-metrics backend (CPU/mem/disk/net), a hard dependency.
#  IMPORTANT: use dgop-bin, NOT dgop-git. The git PKGBUILD builds ./cmd/cli,
#  which upstream renamed to cmd/dgop, so dgop-git fails with
#  "stat .../cmd/cli: directory not found".
dgop-bin

#  accountsservice: hard dependency of dms-shell (user avatar / account info).
accountsservice

# -----------------------------------------------------------------------------
#  Shell runtime dependencies
# -----------------------------------------------------------------------------
#  Wallpaper-driven color scheme generation.
matugen

#  wl-clipboard provides wl-copy/wl-paste for scripts and CLI use.
#  NOTE: cliphist is NOT needed. DMS implements clipboard history itself
#  (IPC target `clipboard`: open/close/toggle, bound to SUPER+v in DMS's own
#  binds fragment), so running cliphist too means two processes racing to own
#  the same Wayland selection. Only add it if you want an external picker.
wl-clipboard

#  Screenshots: DMS has a built-in screenshot service (`dms screenshot`,
#  bound to the Print keys in its binds fragment), so grim/slurp are not
#  required for those. Kept because they are tiny and many scripts and other
#  tools still expect them.
grim
slurp
#  Colour picker. DMS has one too (IPC target `color-picker`), but hyprpicker
#  is kept as the external CLI alternative used by SUPER+SHIFT+C.
hyprpicker

#  Brightness: brightnessctl drives the laptop panel (eDP-1) via sysfs.
#
#  ddcutil drives EXTERNAL monitors over I2C/DDC, but on THIS machine
#  DDC/CI does not work (verified 2026-08-03, ddcutil 2.2.7):
#    DP-2     (BenQ GW2381) -> "DDC communication failed", every attempt,
#                              including --sleep-multiplier 5.
#    HDMI-A-1 (BenQ GW2381) -> detected in ~4 of 5 passes, reads sometimes
#                              return brightness, but every setvcp WRITE
#                              failed (DDCRC_RETRIES / DDCRC_READ_ALL_ZERO).
#  Kept installed anyway: it is the only way to retry after enabling the
#  DDC/CI option in each BenQ's OSD menu (ships disabled on many units).
#  DMS reads brightness backends itself; ddcutil stays installed so external
#  monitor control starts working the moment DDC/CI is enabled in the OSD.
#  Retest with: ddcutil detect && ddcutil -d 1 setvcp 10 80
ddcutil
brightnessctl
i2c-tools

#  noto-fonts-emoji supplies NotoColorEmoji.ttf (CBDT/CBLC color bitmaps).
#  Installing it is NOT sufficient by itself: other fonts that also claim
#  emoji code points (e.g. otf-font-awesome, or any Segoe UI Symbol/Emoji
#  files dropped into ~/.fonts) can outrank it in fontconfig's
#  coverage-sorted fallback, rendering emoji monochrome instead of color.
#  configs/.config/fontconfig/conf.d/76-noto-color-emoji.conf strips the
#  emoji ranges from known offenders at scan time so Noto wins regardless
#  of what else is installed. See that file.
