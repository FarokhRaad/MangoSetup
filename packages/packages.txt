# =============================================================================
#  Package manifest for mango + Noctalia Shell
# =============================================================================
#  Format:  one package per line. '#' starts a comment.
#           Lines ending in "  # optional" are skipped unless --with-optional
#           is passed to 10-packages.sh.
#
#  The installer routes each package to pacman or the AUR automatically and
#  SKIPS anything already installed, so this is safe to re-run.
#
#  Packages the old manifest listed that are intentionally gone:
#    dms-shell-bin           replaced by Noctalia
#    dgop                    DankMaterialShell system monitor backend
#    greetd-dms-greeter-git  replaced by greetd + tuigreet
#    quickshell / quickshell-git
#                            NOT NEEDED. Noctalia v5 is native C++ and dropped
#                            Quickshell entirely. Only v4.x required it.
#    noctalia-shell          that package is Noctalia v4.7.7 (Quickshell/QML).
#                            v5 is the package 'noctalia'. See below.
#    mangowc / mangowc-git   RENAMED. The project moved to the 'mangowm' org
#                            and both old AUR names now 404.
#    qt5ct-kde / qt6ct-kde   the plain qt5ct/qt6ct packages are in [extra]
#    xdg-users-dirs          typo; the real package is xdg-user-dirs
#    breeze5                 Plasma 5 leftover, not needed
# =============================================================================

# -----------------------------------------------------------------------------
#  Shell (Noctalia v5)
# -----------------------------------------------------------------------------
#  Noctalia v5 is a native C++/Wayland/OpenGL-ES rewrite. It has NO Qt, NO GTK
#  and NO Quickshell dependency. Config is TOML, IPC is `noctalia msg`.
#
#  AUR names verified 2026-07-30:
#    noctalia            5.0.0_beta.6-2   <- v5, used here
#    noctalia-git        5.0.0.r1191...   v5 tracking main
#    noctalia-shell      4.7.7-3          OLD v4 Quickshell/QML generation
#
#  v5 is in beta. If you want to pin to the tested beta rather than follow
#  main, keep 'noctalia'; switch to 'noctalia-git' to track development.
noctalia

#  Wallpaper-driven color scheme generation.
matugen

#  wl-clipboard provides wl-copy/wl-paste for scripts and CLI use.
#  NOTE: cliphist is NOT needed on Noctalia v5. v5 implements clipboard
#  history internally, so running cliphist too means two processes racing to
#  own the same Wayland selection. Only add it if you disable
#  [shell] clipboard_enabled and want an external picker.
wl-clipboard

#  Screenshots: v5 has a built-in screenshot service, so grim/slurp are not
#  required for the Print bindings. Kept because they are tiny and many
#  scripts and other tools still expect them.
grim
slurp
#  Colour picker: no Noctalia equivalent, still needed for SUPER+SHIFT+C.
hyprpicker

#  Brightness: brightnessctl drives the laptop panel (eDP-1) via sysfs.
#
#  ddcutil drives EXTERNAL monitors over I2C/DDC, but on THIS machine
#  DDC/CI does not work (verified 2026-08-03, ddcutil 2.2.7):
#    DP-2     (BenQ GW2381) -> "DDC communication failed", every attempt,
#                              including --sleep-multiplier 5.
#    HDMI-A-1 (BenQ GW2381) -> detected in ~4 of 5 passes, reads sometimes
#                              return brightness, but every setvcp WRITE
#                              failed (DDCRC_RETRIES / DDCRC_READ_ALL_ZERO).
#  Kept installed anyway: it is the only way to retry after enabling the
#  DDC/CI option in each BenQ's OSD menu (ships disabled on many units).
#  Noctalia's config.toml sets enable_ddcutil = false to match reality;
#  flip it back to true if `ddcutil -d 1 setvcp 10 80` ever succeeds.
ddcutil
brightnessctl
i2c-tools

#  Audio (PipeWire stack). Noctalia talks to WirePlumber for volume and
#  per-app streams.
pipewire
pipewire-pulse
pipewire-alsa
wireplumber
