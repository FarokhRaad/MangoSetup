# =============================================================================
#  Package manifest for mango + DankMaterialShell
# =============================================================================
#  Format:  one package per line. '#' starts a comment.
#           Lines ending in "  # optional" are skipped unless --with-optional
#           is passed to 10-packages.sh.
#
#  The installer routes each package to pacman or the AUR automatically and
#  skips anything already present, so this is safe to re-run.
#
#  Upstream naming traps worth knowing (all verified against pacman -Si and
#  the AUR RPC, since a wrong name here fails the whole install):
#    mangowc / mangowc-git   RENAMED. The project moved to the 'mangowm' org
#                            and both old AUR names now 404. Use 'mangowm'.
#    p7zip                   RENAMED upstream to '7zip'.
#    qt5ct / qt6ct           the '-kde' variants are used here instead; see
#                            the Qt theming section for why (they conflict,
#                            so do not list both).
#    xdg-users-dirs          common typo; the real package is xdg-user-dirs.
#    dgop-git                its PKGBUILD builds ./cmd/cli, but upstream
#                            renamed that path to cmd/dgop, so the git
#                            package FAILS to build. Use 'dgop-bin'.
# =============================================================================

# -----------------------------------------------------------------------------
#  Prerequisites (NOT in Arch's "base" metapackage)
# -----------------------------------------------------------------------------
#  Arch's `base` provides bash/coreutils/pacman/systemd/pciutils/iproute2 but
#  NOT these. Every one is load-bearing for this setup, and the failure modes
#  are silent, so they are listed explicitly and checked by the scripts:
#    sudo        every script that touches the system uses it
#    git         cloning the mango source for validate-config.sh; AUR builds
#    python      validate-config.sh's parser-inventory checker
#    base-devel  makepkg, required to build ANY AUR package (including yay)
#  If these are missing the scripts refuse to run and tell you this exact
#  command, rather than half-installing a desktop.
sudo
git
python
base-devel

wlr-randr

# -----------------------------------------------------------------------------
#  Shell (DankMaterialShell)
# -----------------------------------------------------------------------------
#  DMS is a Quickshell (QML) + Go desktop shell. It has genuine first-class
#  mango support, not a generic wlroots fallback: it ships a MangoService
#  that speaks mango's native JSON-over-Unix-socket IPC, and `dms setup
#  <fragment>` generates mango config fragments into ~/.config/mango/dms/
#  (colors, layout, cursor, outputs, windowrules, binds), which config.conf
#  sources last. Docs: https://danklinux.com/docs/
#
#  Package choice:
#    dms-shell       stable, in the official [extra] repo
#    dms-shell-git   AUR, tracks main
#  Listed as the git package to stay current with mango IPC changes; switch
#  to 'dms-shell' from [extra] if you prefer stable.
dms-shell-git

#  Quickshell is DMS's ONLY hard UI dependency (in [extra], no AUR fork
#  needed). DMS embeds its QML in the dms binary but still needs the runtime.
quickshell

#  dgop is DMS's system-metrics backend (CPU/mem/disk/net), a hard dependency.
#  It is now in the official [extra] repo, so no AUR package is needed.
#  (Historical note: the old AUR 'dgop-git' failed to build because its
#  PKGBUILD built ./cmd/cli, a path upstream renamed to cmd/dgop. Both the
#  bare 'dgop' AUR package and that workaround are obsolete now that
#  extra/dgop exists.)
dgop

#  accountsservice: hard dependency of dms-shell (user avatar / account info).
accountsservice

#  spm-arch: "Simple Package Manager", an fzf TUI over pacman/yay. NOT an AUR
#  helper itself, it depends on yay and shells out to it (10-packages.sh is what
#  actually bootstraps yay). Ships /usr/bin/spm, a spm_updates.timer that
#  refreshes the package cache, and a libalpm hook.
#
#  IMPORTANT, and the reason this is not just another package line: /usr/bin/spm
#  is a #!/bin/bash script that drives fzf, and fzf executes its --bind and
#  --preview strings through $SHELL. Under a non-bash login shell those bindings
#  (which use bash test syntax such as [[ $w =~ ^[0-9]+$ ]]) misbehave, so
#  configs/.config/zshrc/25-aliases ships:
#      alias spm='SHELL=/bin/bash /usr/bin/spm'
#  Keep the package and that alias together: installing one without the other
#  is what makes this look broken.
spm-arch

#  ffmpegthumbs: video thumbnails in Dolphin. Needed explicitly because it is
#  NOT a dolphin dependency, unlike the other 15 thumbnailers named in
#  configs/.config/dolphinrc, which all ship in kio-extras and arrive with
#  dolphin automatically (verified with `pacman -F` against

ffmpegthumbs
